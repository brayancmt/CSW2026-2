Função em Python para resolver uma equação do segundo grau

            import cmath

            def equacao_segundo_grau(a, b, c):
                if a == 0:
                    raise ValueError("O coeficiente 'a' não pode ser zero (não seria do 2º grau).")
    
            delta = b**2 - 4*a*c
    
            if delta > 0:
                x1 = (-b + delta**0.5) / (2*a)
                x2 = (-b - delta**0.5) / (2*a)
                return x1, x2
            elif delta == 0:
                x = -b / (2*a)
                return x,
            else:
                # raízes complexas
                x1 = (-b + cmath.sqrt(delta)) / (2*a)
                x2 = (-b - cmath.sqrt(delta)) / (2*a)
                return x1, x2
            raizes = equacao_segundo_grau(a, b, c)
            print(f"As raízes são: {raizes}")
